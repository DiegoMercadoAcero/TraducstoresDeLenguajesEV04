function esPar(numero) {
    retornar numero % 2 == 0;
}

function calcular(numero) {
    let resultado = 0;
    si (esPar(numero)) {
        resultado = numero * 2;
    } sino {
        resultado = numero + 1;
    }
    retornar resultado;
}

function proceso() {
    para (i = 0; i < 10; i = i + 1) {
        let res = calcular(i);
        escribir(res);
    }
}
